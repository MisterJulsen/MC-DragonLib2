package de.mrjulsen.mcdragonlib.client.newgui.widgets.richtext.plugin;

import de.mrjulsen.mcdragonlib.client.newgui.widgets.richtext.LineMarker;
import de.mrjulsen.mcdragonlib.client.newgui.widgets.richtext.TextStyle;
import de.mrjulsen.mcdragonlib.client.newgui.widgets.util.TextCursorPosition;
import de.mrjulsen.mcdragonlib.client.util.Graphics;
import de.mrjulsen.mcdragonlib.client.util.GuiUtils; // Assuming GuiUtils has static drawing methods
import de.mrjulsen.mcdragonlib.core.EAlignment;
import net.minecraft.ChatFormatting;
import net.minecraft.client.Minecraft;
import net.minecraft.client.gui.Font;
import net.minecraft.network.chat.Component;
import org.lwjgl.glfw.GLFW;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.stream.Collectors;

public class SuggestionsPlugin {

    private DLRichTextPluginContext context;
    private List<String> allSuggestions = new ArrayList<>();
    private List<String> currentDisplaySuggestions = new ArrayList<>();
    private int selectedSuggestionIndex = -1;
    private String autocompleteGhostText = "";
    private boolean showSuggestionsBox = false;
    private String activePrefix = "";

    private static final int AUTOCOMPLETE_GHOST_TEXT_COLOR = 0xFFA0A0A0; // Grey
    private static final int SUGGESTIONS_BOX_BACKGROUND_COLOR = 0xE0000000; // Semi-transparent black
    private static final int SUGGESTIONS_BOX_BORDER_COLOR = 0xFF888888; // Light Grey
    private static final int SUGGESTIONS_TEXT_COLOR = 0xFFFFFFFF; // White
    private static final int SUGGESTIONS_SELECTED_TEXT_COLOR = 0xFFFFFF00; // Yellow
    private static final int SUGGESTIONS_SELECTED_BACKGROUND_COLOR = 0xFF555555; // Dark Grey

    // Store last calculated component-relative position for mouse down checks
    private float lastPopupDrawComponentX = 0;
    private float lastPopupDrawComponentY = 0;
    private int lastPopupDrawWidth = 0;
    private int lastPopupDrawHeight = 0;


    public SuggestionsPlugin() {
        // Standard-Vorschlagsliste
        this.allSuggestions.addAll(Arrays.asList("Hallo", "Welt", "Minecraft", "Autovervollständigung", "Test", "Beispiel", "Salzingen", "Salzbrot", "Sal"));
    }

    public void setMasterSuggestionList(List<String> suggestions) {
        this.allSuggestions = new ArrayList<>(suggestions);
        clearAutocompleteState();
        if (context != null && context.isFocused()) {
            updateAutocompleteSuggestions();
        }
    }

    public void onAttached(DLRichTextPluginContext context) {
        this.context = context;
    }

    public void onDetached() {
        this.context = null;
        clearAutocompleteState();
    }

    
    public void onCharTyped(DLRichTextPluginCharEvent event) {
        if (context == null || event.isConsumed()) return;
        // Die Logik zur Aktualisierung von Vorschlägen wird hauptsächlich in onTextChanged und onCursorMoved behandelt.
        // Hier könnte man spezifische Logik für Zeicheneingaben hinzufügen, falls nötig.
    }

    
    public void onKeyPressed(DLRichTextPluginKeyEvent event) {
        if (context == null) { // Kontextprüfung am Anfang
            if (showSuggestionsBox && event.getKeyCode() == GLFW.GLFW_KEY_ESCAPE) { // Erlaube Escape auch ohne Kontext
                clearAutocompleteState();
                event.consume();
            }
            return;
        }
        if (event.isConsumed()) return;


        if (showSuggestionsBox && event.getKeyCode() == GLFW.GLFW_KEY_ESCAPE) {
            clearAutocompleteState();
            event.consume();
            return;
        }

        if (!context.isFocused()) {
            return;
        }

        if (showSuggestionsBox) {
            boolean handledByPlugin = false;
            switch (event.getKeyCode()) {
                case GLFW.GLFW_KEY_TAB:
                case GLFW.GLFW_KEY_ENTER:
                case GLFW.GLFW_KEY_KP_ENTER:
                    if (selectedSuggestionIndex != -1 && !currentDisplaySuggestions.isEmpty()) {
                        applySelectedSuggestion();
                        handledByPlugin = true;
                    } else {
                        // Wenn keine Auswahl, aber Ghost-Text vorhanden ist, könnte man den Ghost-Text anwenden
                        // oder einfach den Autocomplete-Status löschen.
                        clearAutocompleteState();
                    }
                    break;
                case GLFW.GLFW_KEY_UP:
                    if (!currentDisplaySuggestions.isEmpty()) {
                        selectedSuggestionIndex--;
                        if (selectedSuggestionIndex < 0) {
                            selectedSuggestionIndex = currentDisplaySuggestions.size() - 1;
                        }
                        updateGhostText();
                        handledByPlugin = true;
                    }
                    break;
                case GLFW.GLFW_KEY_DOWN:
                    if (!currentDisplaySuggestions.isEmpty()) {
                        selectedSuggestionIndex++;
                        if (selectedSuggestionIndex >= currentDisplaySuggestions.size()) {
                            selectedSuggestionIndex = 0;
                        }
                        updateGhostText();
                        handledByPlugin = true;
                    }
                    break;
            }
            if (handledByPlugin) {
                event.consume();
            }
        }
    }

    
    public void onTextChanged() {
        if (context == null) {
            return;
        }
        updateAutocompleteSuggestions();
    }

    
    public void onCursorMoved(boolean textModified) {
        if (context == null) {
            return;
        }
        // Nur aktualisieren, wenn sich der Präfix tatsächlich geändert hat oder die Box nicht sichtbar ist
        // um unnötige Neuberechnungen bei reiner Cursorbewegung innerhalb des gleichen Wortes zu vermeiden.
        String newPrefix = context.getCurrentWordPrefix();
        if (!showSuggestionsBox || (newPrefix != null && !newPrefix.equals(this.activePrefix))) {
            updateAutocompleteSuggestions();
        } else if (newPrefix == null || newPrefix.isEmpty()) {
            clearAutocompleteState();
        }
    }

    
    public void onFocusChanged(boolean focused) {
        if (!focused) {
            clearAutocompleteState();
        } else if (context != null) { // Sicherstellen, dass der Kontext vorhanden ist
            updateAutocompleteSuggestions();
        }
    }

    
    public void onRenderBackground(Graphics graphics, int mouseX, int mouseY) {

    }

    private void clearAutocompleteState() {
        boolean wasShowing = this.showSuggestionsBox;
        this.showSuggestionsBox = false;
        this.currentDisplaySuggestions.clear();
        this.selectedSuggestionIndex = -1;
        this.autocompleteGhostText = "";
        this.activePrefix = ""; // Wichtig, den aktiven Präfix zurückzusetzen
        if (wasShowing && context != null) {
            context.requestRenderUpdate();
        }
    }

    private void updateAutocompleteSuggestions() {
        if (context == null) return;

        String currentCalculatedPrefix = context.getCurrentWordPrefix();
        this.activePrefix = currentCalculatedPrefix; // activePrefix hier setzen

        if (this.activePrefix == null || this.activePrefix.isEmpty()) {
            clearAutocompleteState();
            return;
        }

        String lowerCaseActivePrefix = this.activePrefix.toLowerCase();
        List<String> previousSuggestions = new ArrayList<>(this.currentDisplaySuggestions);
        int previouslySelectedIndex = this.selectedSuggestionIndex;
        String previouslySelectedValue = null;
        if(previouslySelectedIndex >= 0 && previouslySelectedIndex < previousSuggestions.size()){
            previouslySelectedValue = previousSuggestions.get(previouslySelectedIndex);
        }


        this.currentDisplaySuggestions = this.allSuggestions.stream()
                .filter(s -> s.toLowerCase().startsWith(lowerCaseActivePrefix) && !s.equalsIgnoreCase(this.activePrefix))
                .collect(Collectors.toList());

        if (!this.currentDisplaySuggestions.isEmpty()) {
            this.showSuggestionsBox = true;
            // Versuchen, die vorherige Auswahl beizubehalten, wenn der Wert noch vorhanden ist
            int newSelectionIndex = -1;
            if(previouslySelectedValue != null){
                newSelectionIndex = this.currentDisplaySuggestions.indexOf(previouslySelectedValue);
            }

            if (newSelectionIndex != -1) {
                this.selectedSuggestionIndex = newSelectionIndex;
            } else if (!this.currentDisplaySuggestions.isEmpty()){ // Standardmäßig den ersten Eintrag auswählen
                this.selectedSuggestionIndex = 0;
            } else {
                this.selectedSuggestionIndex = -1; // Keine Auswahl, wenn leer
            }
            updateGhostText();
        } else {
            clearAutocompleteState();
        }
        context.requestRenderUpdate(); // UI-Update anfordern
    }

    private void updateGhostText() {
        this.autocompleteGhostText = ""; // Zurücksetzen
        if (this.showSuggestionsBox &&
                this.selectedSuggestionIndex >= 0 &&
                this.selectedSuggestionIndex < this.currentDisplaySuggestions.size()) {

            String fullSuggestion = this.currentDisplaySuggestions.get(this.selectedSuggestionIndex);
            // Sicherstellen, dass activePrefix nicht null ist, bevor auf ihm Methoden aufgerufen werden
            if (this.activePrefix != null && !this.activePrefix.isEmpty() &&
                    fullSuggestion.toLowerCase().startsWith(this.activePrefix.toLowerCase())) {
                // Ghost-Text ist der Teil der vollen Suggestion, der nach dem activePrefix kommt
                if (this.activePrefix.length() < fullSuggestion.length()) {
                    this.autocompleteGhostText = fullSuggestion.substring(this.activePrefix.length());
                }
            }
        }
    }

    private void applySelectedSuggestion() {
        if (context == null || !this.showSuggestionsBox || this.selectedSuggestionIndex < 0 ||
                this.selectedSuggestionIndex >= this.currentDisplaySuggestions.size()) {
            clearAutocompleteState();
            return;
        }

        // activePrefix sollte bereits durch updateAutocompleteSuggestions korrekt gesetzt sein.
        // Falls nicht (z.B. wenn der Cursor bewegt wurde, ohne dass onCursorMoved den Prefix änderte),
        // hole ihn hier frisch.
        String prefixToReplace = context.getCurrentWordPrefix();
        if (prefixToReplace == null || prefixToReplace.isEmpty()) {
            // Wenn der aktuelle Prefix leer ist, aber eine Suggestion ausgewählt wurde,
            // könnte dies ein Sonderfall sein (z.B. Suggestion ohne Tippen anzeigen).
            // Für Standard-Autovervollständigung sollte der Prefix nicht leer sein.
            prefixToReplace = this.activePrefix; // Fallback auf den gespeicherten activePrefix
            if (prefixToReplace == null || prefixToReplace.isEmpty()) {
                clearAutocompleteState();
                return;
            }
        }


        String suggestionToApply = this.currentDisplaySuggestions.get(this.selectedSuggestionIndex);

        int cursorIdx = context.getCursorIndex();
        // Die Länge des zu ersetzenden Präfixes in CodePoints.
        int prefixCpLength = prefixToReplace.codePointCount(0, prefixToReplace.length());
        int prefixStartIndex = cursorIdx - prefixCpLength;

        if (prefixStartIndex < 0) { // Sollte nicht passieren, wenn prefixToReplace korrekt ist
            clearAutocompleteState();
            return;
        }

        // Überprüfen, ob der Text vor dem Cursor tatsächlich dem prefixToReplace entspricht
        String actualTextBeforeCursor = context.getRichText().subComponent(prefixStartIndex, cursorIdx).plainText();
        if (!actualTextBeforeCursor.equals(prefixToReplace)) {
            // Diskrepanz. Möglicherweise hat sich der Text geändert oder der Cursor wurde bewegt.
            // Vorsichtshalber nicht anwenden oder eine robustere Logik implementieren.
            // Fürs Erste: Zustand zurücksetzen.
            clearAutocompleteState();
            return;
        }


        TextStyle styleOfPrefix = context.getStyleAt(Math.max(0, prefixStartIndex)); // Stil am Anfang des Präfixes holen

        // Verwende replaceText, um den Präfix durch die Suggestion zu ersetzen
        context.replaceText(prefixStartIndex, cursorIdx, suggestionToApply, styleOfPrefix);
        // replaceText sollte den Cursor bereits an das Ende des eingefügten Textes setzen
        // und onTextChanged auslösen, was wiederum updateAutocompleteSuggestions aufruft.

        clearAutocompleteState(); // Zustand nach Anwendung zurücksetzen
    }


    
    public void onRenderOverlay(de.mrjulsen.mcdragonlib.client.util.Graphics graphics, int mouseX, int mouseY) {
        if (context == null) return;
        Font font = context.getFont();

        // --- Ghost-Text rendern ---
        if (context.isFocused() && showSuggestionsBox && this.autocompleteGhostText != null && !this.autocompleteGhostText.isEmpty() &&
                this.activePrefix != null && !this.activePrefix.isEmpty()) { // Sicherstellen, dass activePrefix nicht null ist
            TextCursorPosition cursorPos = context.getPosByIndex(context.getCursorIndex()); // In Textblock-Koordinaten
            if (cursorPos != null) {
                // Konvertiere Cursorposition in Komponenten-relative Koordinaten
                float ghostDrawX = context.getLayoutContentX() + cursorPos.x() - context.getScrollOffsetX();
                float ghostDrawY = context.getLayoutContentY() + cursorPos.y() - context.getScrollOffsetY();

                // Clipping-Grenzen (Komponenten-relativ)
                float clipLeft = context.getLayoutContentX();
                float clipTop = context.getLayoutContentY();
                float clipRight = clipLeft + context.getLayoutContentWidth();
                float clipBottom = clipTop + context.getLayoutContentHeight();

                // Prüfen, ob der Ghost-Text vertikal innerhalb des geclippten Layout-Inhaltsbereichs liegt
                if (ghostDrawY + cursorPos.lineHeight() > clipTop && ghostDrawY < clipBottom) {
                    // Maximal verfügbare Breite für Ghost-Text innerhalb des Layout-Bereichs der Komponente
                    float availableWidthForGhost = clipRight - ghostDrawX;
                    if (availableWidthForGhost > 0) {
                        String ghostPartToRender = font.plainSubstrByWidth(this.autocompleteGhostText, (int) availableWidthForGhost);
                        if (!ghostPartToRender.isEmpty()) {
                            // GuiUtils zeichnet relativ zum aktuellen PoseStack (der am Ursprung der Komponente liegt)
                            GuiUtils.drawString(graphics, font, (int) ghostDrawX, (int) ghostDrawY, ghostPartToRender, AUTOCOMPLETE_GHOST_TEXT_COLOR, EAlignment.LEFT, false);
                        }
                    }
                }
            }
        }

        // --- Suggestions-Box rendern ---
        if (showSuggestionsBox && !currentDisplaySuggestions.isEmpty()) {
            TextCursorPosition cursorPos = context.getPosByIndex(context.getCursorIndex()); // In Textblock-Koordinaten
            if (cursorPos == null) return;

            LineMarker currentLine = context.getLineAt(context.getCursorIndex()); // In Textblock-Koordinaten
            if (currentLine == null) return;

            // Initiale Komponenten-relative X-Position für die Box (ausgerichtet am Cursor)
            float initialBoxDrawX = context.getLayoutContentX() + cursorPos.x() - context.getScrollOffsetX();
            // Initiale Komponenten-relative Y-Position für die Box (unterhalb der aktuellen Zeile)
            float initialBoxDrawY = context.getLayoutContentY() + (currentLine.y() + currentLine.lineHeight() + context.getLineSpacing() / 2.0f) - context.getScrollOffsetY();

            int maxTextWidth = 0;
            for (String suggestion : currentDisplaySuggestions) {
                maxTextWidth = Math.max(maxTextWidth, font.width(suggestion));
            }

            int padding = 3;
            int itemHeight = font.lineHeight + 1; // +1 für kleinen Abstand zwischen Items
            int boxWidth = maxTextWidth + padding * 2;
            int boxHeight = currentDisplaySuggestions.size() * itemHeight + padding * 2;

            // Bildschirmüberlaufprüfung (verwendet absolute Bildschirmkoordinaten)
            Minecraft mc = Minecraft.getInstance();
            int screenWidth = mc.getWindow().getGuiScaledWidth();
            int screenHeight = mc.getWindow().getGuiScaledHeight();

            // Absolute Bildschirmposition der Box, wenn sie bei initialComponentX/Y gezeichnet würde
            float absBoxScreenX = context.getComponentX() + initialBoxDrawX;
            float absBoxScreenY = context.getComponentY() + initialBoxDrawY;

            // Absolute Bildschirmposition anpassen, wenn sie überläuft
            if (absBoxScreenX + boxWidth > screenWidth) {
                absBoxScreenX = screenWidth - boxWidth - 5; // Rand zum Bildschirmrand
            }
            if (absBoxScreenY + boxHeight > screenHeight) {
                // Versuche, die Box über der Zeile zu positionieren
                float lineTopComponentY = context.getLayoutContentY() + (currentLine.y() - context.getLineSpacing() / 2.0f) - context.getScrollOffsetY();
                absBoxScreenY = (context.getComponentY() + lineTopComponentY) - boxHeight;
            }
            absBoxScreenX = Math.max(0, absBoxScreenX); // An linkem Bildschirmrand festklemmen
            absBoxScreenY = Math.max(0, absBoxScreenY); // An oberem Bildschirmrand festklemmen

            // Finale absolute Bildschirmposition zurück in Komponenten-relative Koordinaten für das Zeichnen umwandeln
            float finalBoxDrawComponentX = absBoxScreenX - context.getComponentX();
            float finalBoxDrawComponentY = absBoxScreenY - context.getComponentY();

            // Speichern für Mausinteraktion
            this.lastPopupDrawComponentX = finalBoxDrawComponentX;
            this.lastPopupDrawComponentY = finalBoxDrawComponentY;
            this.lastPopupDrawWidth = boxWidth;
            this.lastPopupDrawHeight = boxHeight;


            // Hintergrund und Rand zeichnen (mit Komponenten-relativen Koordinaten)
            GuiUtils.fill(graphics, (int)finalBoxDrawComponentX - 1, (int)finalBoxDrawComponentY - 1, boxWidth + 2, boxHeight + 2, SUGGESTIONS_BOX_BORDER_COLOR);
            GuiUtils.fill(graphics, (int)finalBoxDrawComponentX, (int)finalBoxDrawComponentY, boxWidth, boxHeight, SUGGESTIONS_BOX_BACKGROUND_COLOR);

            // Vorschläge zeichnen (mit Komponenten-relativen Koordinaten)
            float currentItemDrawY = finalBoxDrawComponentY + padding;
            for (int i = 0; i < currentDisplaySuggestions.size(); i++) {
                String suggestion = currentDisplaySuggestions.get(i);
                int textColor = (i == selectedSuggestionIndex) ? SUGGESTIONS_SELECTED_TEXT_COLOR : SUGGESTIONS_TEXT_COLOR;
                if (i == selectedSuggestionIndex) {
                    GuiUtils.fill(graphics, (int)finalBoxDrawComponentX + 1, (int)currentItemDrawY -1, boxWidth - 2, itemHeight, SUGGESTIONS_SELECTED_BACKGROUND_COLOR);
                }
                GuiUtils.drawString(graphics, font, (int)finalBoxDrawComponentX + padding, (int)currentItemDrawY, suggestion, textColor, EAlignment.LEFT, false);
                currentItemDrawY += itemHeight;
            }
        }
    }

    
    public void onMouseDown(double mouseX, double mouseY, int button, DLRichTextPluginEvent event) {
        // mouseX, mouseY sind bereits Komponenten-relativ
        if (showSuggestionsBox && context != null && !currentDisplaySuggestions.isEmpty()) {
            // Prüfen, ob der Klick innerhalb der zuletzt gerenderten Popup-Grenzen liegt
            if (mouseX >= lastPopupDrawComponentX && mouseX <= lastPopupDrawComponentX + lastPopupDrawWidth &&
                    mouseY >= lastPopupDrawComponentY && mouseY <= lastPopupDrawComponentY + lastPopupDrawHeight) {

                Font font = context.getFont();
                int padding = 3;
                int itemHeight = font.lineHeight + 1;
                // Y-Position des Mausklicks relativ zum Beginn des Item-Bereichs im Popup
                float relativeYInBoxItems = (float)mouseY - (lastPopupDrawComponentY + padding);
                int clickedIndex = (int) (relativeYInBoxItems / itemHeight);

                if (clickedIndex >= 0 && clickedIndex < currentDisplaySuggestions.size()) {
                    selectedSuggestionIndex = clickedIndex;
                    applySelectedSuggestion();
                    event.consume(); // Ereignis konsumieren, da es vom Plugin behandelt wurde
                }
            } else {
                // Außerhalb der aktuell angezeigten Suggestion-Box geklickt
                clearAutocompleteState();
                // Nicht konsumieren, damit der Editor den Klick verarbeiten kann.
            }
        }
    }
    
    public void onMouseUp(double mouseX, double mouseY, int button, DLRichTextPluginEvent event) {}
    
    public void onMouseDragged(double mouseX, double mouseY, int button, double dragX, double dragY, DLRichTextPluginEvent event) {}
    
    public void onScroll(double mouseX, double mouseY, double scrollX, double scrollY, DLRichTextPluginEvent event) {}
    
    public void tick() {}
}
